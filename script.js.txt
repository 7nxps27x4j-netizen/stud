// [수정] 한국 시간(KST) 기준으로 날짜 문자열을 반환하는 함수
function getKSTDateString() {
    const now = new Date();
    // 한국 시간은 UTC+9 이므로 오프셋을 조절하여 날짜 추출
    const kstOffset = 9 * 60 * 60 * 1000;
    const kstDate = new Date(now.getTime() + kstOffset);
    return kstDate.toISOString().split('T')[0];
}

function checkDateChange() {
    const nowStr = getKSTDateString();
    if (nowStr !== currentSystemDate) {
        currentSystemDate = nowStr;
        render2026Calendar();
        updateUI();
    }
}

function getDayTotal(dateStr) {
    if (!studyData[dateStr]) return 0;
    return Object.values(studyData[dateStr]).reduce((a, b) => a + b, 0);
}

function updateUI() {
    const today = getKSTDateString(); // [수정] KST 기준 호출
    const todayLogs = studyData[today] || {};
    
    const todayTotal = getDayTotal(today);
    document.getElementById('todayTime').innerText = formatTime(todayTotal);
    
    const currentYearMonth = today.substring(0, 7);
    let monthTotalSec = 0;
    Object.keys(studyData).forEach(date => {
        if (date.startsWith(currentYearMonth)) {
            monthTotalSec += getDayTotal(date);
        }
    });
    document.getElementById('monthTime').innerText = formatTime(monthTotalSec);
    
    // 스트릭 로직 수정: KST 기준으로 오늘부터 역산
    let streak = 0;
    let checkDate = new Date();
    // 루프 돌 때마다 한국 시간 오프셋 적용
    while (true) {
        const kstOffset = 9 * 60 * 60 * 1000;
        const kstCheck = new Date(checkDate.getTime() + kstOffset);
        let dateStr = kstCheck.toISOString().split('T')[0];
        
        if (getDayTotal(dateStr) > 0) {
            streak++;
            checkDate.setDate(checkDate.getDate() - 1);
        } else {
            break;
        }
    }
    document.getElementById('streakCount').innerText = `${streak}일`;

    const listContainer = document.getElementById('subjectTimeList');
    listContainer.innerHTML = '';
    ['국어', '수학', '영어', '과학'].forEach(sub => {
        const li = document.createElement('li');
        const time = todayLogs[sub] || 0;
        li.innerHTML = `<span>${sub}</span> <strong>${formatTime(time)}</strong>`;
        listContainer.appendChild(li);
    });
}

function render2026Calendar() {
    const container = document.getElementById('grassContainer');
    container.innerHTML = '';
    // 달력 연/월 계산 시에도 한국 시간 기준 권장
    const nowKST = new Date(new Date().getTime() + (9*60*60*1000));
    const year = 2026, month = nowKST.getUTCMonth(); 
    
    const firstDayIdx = new Date(year, month, 1).getDay();
    const lastDate = new Date(year, month + 1, 0).getDate();
    
    document.getElementById('monthTitle').innerText = `2026년 ${month + 1}월 공부 기록`;

    for(let i=0; i<firstDayIdx; i++) {
        const empty = document.createElement('div');
        empty.className = 'grass-cell';
        empty.style.opacity = '0';
        container.appendChild(empty);
    }

    for (let i = 1; i <= lastDate; i++) {
        const dateStr = `${year}-${String(month+1).padStart(2,'0')}-${String(i).padStart(2,'0')}`;
        const totalSec = getDayTotal(dateStr);
        
        const cell = document.createElement('div');
        cell.className = 'grass-cell';
        
        if (dateStr === currentSystemDate) {
            cell.style.border = "2px solid #f97316";
            cell.style.boxShadow = "0 0 10px rgba(249, 115, 22, 0.5)";
        }
        
        cell.innerHTML = `<span class="cell-date">${i}</span>`;
        if(totalSec > 0) {
            cell.innerHTML += `<span class="cell-time-text">${formatSimpleTime(totalSec)}</span>`;
            let lv = Math.min(Math.ceil(totalSec / 3600), 5); 
            cell.classList.add(`lv-${lv}`);
        }
        container.appendChild(cell);
    }
}

function formatTime(sec) {
    const h = Math.floor(sec / 3600).toString().padStart(2, '0');
    const m = Math.floor((sec % 3600) / 60).toString().padStart(2, '0');
    const s = (sec % 60).toString().padStart(2, '0');
    return `${h}:${m}:${s}`;
}

function formatSimpleTime(sec) {
    const h = Math.floor(sec / 3600);
    const m = Math.floor((sec % 3600) / 60);
    return `${h}h ${m}m`;
}

document.getElementById('startBtn').addEventListener('click', () => {
    document.getElementById('startBtn').disabled = true;
    document.getElementById('stopBtn').disabled = false;
    timerInterval = setInterval(() => {
        secondsElapsed++;
        document.getElementById('stopwatchDisplay').innerText = formatTime(secondsElapsed);
    }, 1000);
});

document.getElementById('stopBtn').addEventListener('click', () => {
    clearInterval(timerInterval);
    const subject = document.getElementById('subjectSelect').value;
    const today = getKSTDateString(); // [수정] 저장 시에도 한국 시간 기준
    
    if(!studyData[today]) studyData[today] = {};
    studyData[today][subject] = (studyData[today][subject] || 0) + secondsElapsed;
    
    localStorage.setItem(STORAGE_KEY, JSON.stringify(studyData));
    
    secondsElapsed = 0;
    document.getElementById('stopwatchDisplay').innerText = "00:00:00";
    document.getElementById('startBtn').disabled = false;
    document.getElementById('stopBtn').disabled = true;
    
    updateUI();
    render2026Calendar();
});

document.getElementById('themeToggle').addEventListener('click', () => {
    document.body.classList.toggle('dark-mode');
    document.body.classList.toggle('light-mode');
});
